# micro-ROS Agent (Entwicklung: zwei einfache Container)
# Feste Device-Namen per udev-Regel (/etc/udev/rules.d/99-scheerenrobot-esp32.rules,
# siehe idf-micro-ros-setup.md), gebunden an physisch markierte USB-Buchsen am Pi5.
# ttyUSB0/1 NICHT verwenden - beide CP2102-Chips melden dieselbe generische
# USB-Seriennummer "0001", die Nummerierung ist daher nicht enumerationsstabil.
# TODO sobald Autostart nach Power-On gebraucht wird: docker-compose statt
# manueller docker run-Aufrufe.
docker run -it --rm -v /dev:/dev -v /dev/shm:/dev/shm --privileged --net=host \
  --name microros-agent-robot \
  microros/micro-ros-agent:jazzy serial --dev /dev/ttyUSB-robot -b 921600

docker run -it --rm -v /dev:/dev -v /dev/shm:/dev/shm --privileged --net=host \
  --name microros-agent-lift \
